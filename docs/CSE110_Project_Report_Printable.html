1. Project Description & Objectives

Gym Tracker is a Java desktop application with a Java Swing graphical user interface built to manage gym members, certified trainers, daily strength and cardio workout logs, and monthly membership/salary calculations. All data is saved automatically to local text files so records are preserved between sessions.

2. Key Features

  • Member Management: Add, update, delete, and search members, compute BMI automatically, choose Basic/Standard/Premium plans, and assign a personal trainer.
  • Trainer Management: Manage trainers, specialties, base salaries, and automatic commission calculation (+BDT 750 per trainee).
  • Workout Logbook: Log Strength (sets, reps, weight) or Cardio (duration, calories) sessions using overloaded methods and filter logs by member.
  • Summary & Polymorphism View: Combines Member and Trainer objects inside ArrayList<Person> to calculate monthly inflows, payroll, and summaries dynamically.
  • File Persistence: Reads and writes data/members.txt, data/trainers.txt, and data/workouts.txt.

package gym;

import java.util.ArrayList;

// Subclass of Person representing a gym member.
// Demonstrates: Inheritance, Polymorphism (Overriding + Overloading), Association & Composition.
public class Member extends Person {
    private String plan; // "Basic", "Standard", or "Premium"
    private double weightKg;
    private double heightCm;
    private Trainer assignedTrainer; // Association (Aggregation)
    private ArrayList<WorkoutLog> workouts; // Composition

    public Member(String id, String name, String phone, String email,
                  String plan, double weightKg, double heightCm, Trainer assignedTrainer) {
        super(id, name, phone, email);
        this.plan = plan;
        this.weightKg = weightKg;
        this.heightCm = heightCm;
        this.assignedTrainer = assignedTrainer;
        this.workouts = new ArrayList<>();
    }

    public String getPlan() {
        return plan;
    }

    public void setPlan(String plan) {
        this.plan = plan;
    }

    public double getWeightKg() {
        return weightKg;
    }

    public void setWeightKg(double weightKg) {
        this.weightKg = weightKg;
    }

    public double getHeightCm() {
        return heightCm;
    }

    public void setHeightCm(double heightCm) {
        this.heightCm = heightCm;
    }

    public Trainer getAssignedTrainer() {
        return assignedTrainer;
    }

    public void setAssignedTrainer(Trainer assignedTrainer) {
        this.assignedTrainer = assignedTrainer;
    }

    public String getAssignedTrainerId() {
        return assignedTrainer != null ? assignedTrainer.getId() : "NONE";
    }

    public String getAssignedTrainerName() {
        return assignedTrainer != null ? assignedTrainer.getName() : "None";
    }

    public ArrayList<WorkoutLog> getWorkouts() {
        return workouts;
    }

    public double calculateBMI() {
        double hMeters = heightCm / 100.0;
        return weightKg / (hMeters * hMeters);
    }

    public String getBmiStatus() {
        double bmi = calculateBMI();
        if (bmi < 18.5) return "Underweight";
        if (bmi < 25.0) return "Normal";
        if (bmi < 30.0) return "Overweight";
        return "Obese";
    }

    // --- Method Overloading (Compile-Time Polymorphism) ---

    public void addWorkout(WorkoutLog log) {
        workouts.add(log);
    }

    // Overloaded method for Strength workouts
    public WorkoutLog addWorkout(String logId, String date, String exercise,
                                 int sets, int reps, double weightKg) {
        WorkoutLog log = new WorkoutLog(logId, getId(), date, exercise, sets, reps, weightKg);
        workouts.add(log);
        return log;
    }

    // Overloaded method for Cardio workouts
    public WorkoutLog addWorkout(String logId, String date, String exercise,
                                 int durationMinutes, int caloriesBurned) {
        WorkoutLog log = new WorkoutLog(logId, getId(), date, exercise, durationMinutes, caloriesBurned);
        workouts.add(log);
        return log;
    }

    // --- Method Overriding (Runtime Polymorphism) ---

    @Override
    public String getRole() {
        return "Member (" + plan + ")";
    }

    @Override
    public double calculateMonthlyAmount() {
        double fee = 1500.0; // Basic
        if ("Standard".equalsIgnoreCase(plan)) {
            fee = 2500.0;
        } else if ("Premium".equalsIgnoreCase(plan)) {
            fee = 4000.0;
        }
        if (assignedTrainer != null) {
            fee += 1500.0; // Personal trainer addon
        }
        return fee;
    }

    @Override
    public String getSummary() {
        return super.getSummary() + String.format(" | BMI: %.1f (%s) | Monthly Fee: BDT %.0f",
                calculateBMI(), getBmiStatus(), calculateMonthlyAmount());
    }

    public String toFileLine() {
        return getId() + "|" + getName() + "|" + getPhone() + "|" + getEmail() + "|"
                + plan + "|" + weightKg + "|" + heightCm + "|" + getAssignedTrainerId();
    }
}

                   String specialty, double baseSalary, int experienceYears) {
        super(id, name, phone, email);
        this.specialty = specialty;
        this.baseSalary = baseSalary;
        this.experienceYears = experienceYears;
        this.assignedMembers = new ArrayList<>();
    }

    public String getSpecialty() {
        return specialty;
    }

    public void setSpecialty(String specialty) {
        this.specialty = specialty;
    }

    public double getBaseSalary() {
        return baseSalary;
    }

    public void setBaseSalary(double baseSalary) {
        this.baseSalary = baseSalary;
    }

    public int getExperienceYears() {
        return experienceYears;
    }

    public void setExperienceYears(int experienceYears) {
        this.experienceYears = experienceYears;
    }

    public ArrayList<Member> getAssignedMembers() {
        return assignedMembers;
    }

    public void addAssignedMember(Member member) {
        if (member != null && !assignedMembers.contains(member)) {
            assignedMembers.add(member);
        }
    }

    public void clearAssignedMembers() {
        assignedMembers.clear();
    }

    @Override
    public String getRole() {
        return "Trainer (" + specialty + ")";
    }

    @Override
    public double calculateMonthlyAmount() {
        // Base salary + BDT 750 commission per assigned member
        return baseSalary + (assignedMembers.size() * 750.0);
    }

    @Override
    public String getSummary() {
        return super.getSummary() + String.format(" | Exp: %d yrs | Trainees: %d | Salary: BDT %.0f",
                experienceYears, assignedMembers.size(), calculateMonthlyAmount());
    }

    public String toFileLine() {
        return getId() + "|" + getName() + "|" + getPhone() + "|" + getEmail() + "|"
                + specialty + "|" + baseSalary + "|" + experienceYears;
    }

    public static Trainer fromFileLine(String line) {
        String[] p = line.split("\\|");
        return new Trainer(
                p[0], p[1], p[2], p[3], p[4],
                Double.parseDouble(p[5]),
                Integer.parseInt(p[6])
        );
    }
}

package gym;

// Represents a single Strength or Cardio workout logged by a member.
public class WorkoutLog implements Identifiable {
    private String logId;
    private String memberId;
    private String date;
    private String exerciseName;
    private String type; // "Strength" or "Cardio"
    private int sets;
    private int reps;
    private double weightKg;
    private int durationMinutes;
    private int caloriesBurned;

    // Full constructor (used when loading from file)
    public WorkoutLog(String logId, String memberId, String date, String exerciseName,
                      String type, int sets, int reps, double weightKg,
                      int durationMinutes, int caloriesBurned) {
        this.logId = logId;
        this.memberId = memberId;
        this.date = date;
        this.exerciseName = exerciseName;
        this.type = type;
        this.sets = sets;
        this.reps = reps;
        this.weightKg = weightKg;
        this.durationMinutes = durationMinutes;
        this.caloriesBurned = caloriesBurned;
    }

    // Overloaded Constructor 1: Strength workout
    public WorkoutLog(String logId, String memberId, String date, String exerciseName,
                      int sets, int reps, double weightKg) {
        this(logId, memberId, date, exerciseName, "Strength", sets, reps, weightKg,
                sets * 3, (int) Math.round(sets * reps * (weightKg * 0.1 + 1.5)));
    }

    // Overloaded Constructor 2: Cardio workout
    public WorkoutLog(String logId, String memberId, String date, String exerciseName,
                      int durationMinutes, int caloriesBurned) {
        this(logId, memberId, date, exerciseName, "Cardio", 0, 0, 0.0,
                durationMinutes, caloriesBurned);
    }

    @Override
    public String getId() {
        return logId;
    }

    public String getMemberId() {
        return memberId;
    }

    public String getDate() {
        return date;
    }

    public String getExerciseName() {
        return exerciseName;
    }

    public String getType() {
        return type;
    }

    public int getSets() {
        return sets;
    }

    public int getReps() {
        return reps;
    }

    public double getWeightKg() {
        return weightKg;
    }

    public int getDurationMinutes() {
        return durationMinutes;
    }

    public int getCaloriesBurned() {
        return caloriesBurned;
    }

    public double getTotalVolume() {
        return sets * reps * weightKg;
    }

    @Override
    public String getSummary() {
        if ("Strength".equalsIgnoreCase(type)) {
            return date + " | " + exerciseName + " (" + sets + "x" + reps + " @ " + weightKg + " kg)";
        }
        return date + " | " + exerciseName + " (" + durationMinutes + " min, " + caloriesBurned + " kcal)";
    }

    public String toFileLine() {
        return logId + "|" + memberId + "|" + date + "|" + exerciseName + "|" + type + "|"
                + sets + "|" + reps + "|" + weightKg + "|" + durationMinutes + "|" + caloriesBurned;
    }

    public static WorkoutLog fromFileLine(String line) {
        String[] p = line.split("\\|");
        return new WorkoutLog(
                p[0], p[1], p[2], p[3], p[4],
                Integer.parseInt(p[5]),
                Integer.parseInt(p[6]),
                Double.parseDouble(p[7]),
                Integer.parseInt(p[8]),
                Integer.parseInt(p[9])
        );
    }
}

package gym;

import java.util.ArrayList;
import java.util.HashMap;

// Generic repository class demonstrating Generics (<T extends Identifiable>)
// and Java Collections (ArrayList and HashMap).
public class DataManager<T extends Identifiable> {
    private ArrayList<T> list = new ArrayList<>();
    private HashMap<String, T> map = new HashMap<>();

    public void add(T item) throws DuplicateIdException {
        String key = item.getId().trim().toUpperCase();
        if (map.containsKey(key)) {
            throw new DuplicateIdException("ID '" + item.getId() + "' already exists!");
        }
        list.add(item);
        map.put(key, item);
    }

    public T findById(String id) throws NotFoundException {
        String key = id.trim().toUpperCase();
        T item = map.get(key);
        if (item == null) {
            throw new NotFoundException("No record found with ID '" + id + "'.");
        }
        return item;
    }

    public void remove(String id) throws NotFoundException {
        T item = findById(id);
        list.remove(item);
        map.remove(id.trim().toUpperCase());
    }

    public boolean exists(String id) {
        return map.containsKey(id.trim().toUpperCase());
    }

    public ArrayList<T> getAll() {
        return list;
    }

    public void clear() {
        list.clear();
        map.clear();
    }

    public int size() {
        return list.size();
    }
}

package gym;

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.File;
import java.io.FileReader;
import java.io.FileWriter;
import java.io.IOException;
import java.util.HashMap;

// Handles saving and loading text files using BufferedReader and BufferedWriter.
public class FileHandler {
    private File dataFolder;
    private File membersFile;
    private File trainersFile;
    private File workoutsFile;

    public FileHandler(String folderName) {
        this.dataFolder = new File(folderName);
        this.trainersFile = new File(dataFolder, "trainers.txt");
        this.membersFile = new File(dataFolder, "members.txt");
        this.workoutsFile = new File(dataFolder, "workouts.txt");
    }

    public boolean dataFilesExist() {
        return trainersFile.exists() || membersFile.exists() || workoutsFile.exists();
    }

    public void saveAll(DataManager<Trainer> trainers,
                        DataManager<Member> members,
                        DataManager<WorkoutLog> workouts) throws IOException {
        if (!dataFolder.exists()) {
            dataFolder.mkdirs();
        }

        try (BufferedWriter bw = new BufferedWriter(new FileWriter(trainersFile))) {
            for (Trainer t : trainers.getAll()) {
                bw.write(t.toFileLine());
                bw.newLine();
            }
        }

        try (BufferedWriter bw = new BufferedWriter(new FileWriter(membersFile))) {
            for (Member m : members.getAll()) {
                bw.write(m.toFileLine());
                bw.newLine();
            }
        }

        try (BufferedWriter bw = new BufferedWriter(new FileWriter(workoutsFile))) {
            for (WorkoutLog w : workouts.getAll()) {
                bw.write(w.toFileLine());
                bw.newLine();
            }
        }
    }

    public void loadAll(DataManager<Trainer> trainers,
                        DataManager<Member> members,
                        DataManager<WorkoutLog> workouts) throws IOException {
        trainers.clear();
        members.clear();
        workouts.clear();

        HashMap<String, Trainer> trainerMap = new HashMap<>();

        if (trainersFile.exists()) {
            try (BufferedReader br = new BufferedReader(new FileReader(trainersFile))) {
                String line;
                while ((line = br.readLine()) != null) {
                    if (!line.trim().isEmpty()) {
                        Trainer t = Trainer.fromFileLine(line.trim());
                        try {
                            trainers.add(t);
                            trainerMap.put(t.getId().toUpperCase(), t);
                        } catch (DuplicateIdException e) {
                            System.err.println("Skipped duplicate trainer: " + e.getMessage());
                        }
                    }
                }
            }
        }

        if (membersFile.exists()) {
            try (BufferedReader br = new BufferedReader(new FileReader(membersFile))) {
                String line;
                while ((line = br.readLine()) != null) {
                    if (!line.trim().isEmpty()) {
                        String[] p = line.trim().split("\\|");
                        String trainerId = p[7];
                        Trainer assigned = "NONE".equalsIgnoreCase(trainerId)
                                ? null : trainerMap.get(trainerId.toUpperCase());
                        Member m = new Member(
                                p[0], p[1], p[2], p[3], p[4],
                                Double.parseDouble(p[5]),
                                Double.parseDouble(p[6]),
                                assigned
                        );
                        try {
                            members.add(m);
                            if (assigned != null) {
                                assigned.addAssignedMember(m);
                            }
                        } catch (DuplicateIdException e) {
                            System.err.println("Skipped duplicate member: " + e.getMessage());
                        }
                    }
                }
            }
        }

        if (workoutsFile.exists()) {
            try (BufferedReader br = new BufferedReader(new FileReader(workoutsFile))) {
                String line;
                while ((line = br.readLine()) != null) {
                    if (!line.trim().isEmpty()) {
                        WorkoutLog log = WorkoutLog.fromFileLine(line.trim());
                        try {
                            workouts.add(log);
                            if (members.exists(log.getMemberId())) {
                                members.findById(log.getMemberId()).addWorkout(log);
                            }
                        } catch (DuplicateIdException | NotFoundException e) {
                            System.err.println("Skipped workout record: " + e.getMessage());
                        }
                    }
                }
            }
        }
    }
}

        wSetsField.setEnabled(isStrength);

    private void logWorkoutAction() {
        try {
            String memberId = extractId((String) wMemberBox.getSelectedItem());
            Member m = memberManager.findById(memberId);
            String exercise = wExerciseField.getText().trim();
            if (exercise.isEmpty()) {
                throw new InvalidDataException("Exercise name cannot be empty.");
            }
            String logId = generateWorkoutId();
            WorkoutLog log;

            if ("Strength".equals(wTypeBox.getSelectedItem())) {
                int sets = Integer.parseInt(wSetsField.getText().trim());
                int reps = Integer.parseInt(wRepsField.getText().trim());
                double weight = Double.parseDouble(wWeightField.getText().trim());
                if (sets <= 0 || reps <= 0 || weight < 0) {
                    throw new InvalidDataException("Sets and reps must be positive numbers.");
                }
                // Calls Overloaded Method 1 (Strength)
                log = m.addWorkout(logId, wDateField.getText().trim(), exercise, sets, reps, weight);
            } else {
                int duration = Integer.parseInt(wDurationField.getText().trim());
                int calories = Integer.parseInt(wCaloriesField.getText().trim());
                if (duration <= 0 || calories <= 0) {
                    throw new InvalidDataException("Duration and calories must be positive numbers.");
                }
                // Calls Overloaded Method 2 (Cardio)
                log = m.addWorkout(logId, wDateField.getText().trim(), exercise, duration, calories);
            }

            workoutManager.add(log);
            wExerciseField.setText("");
            saveDataQuietly();
            refreshAllTables();
        } catch (NumberFormatException ex) {
            showError("Please enter valid numeric values for workout fields!");
        } catch (InvalidDataException | NotFoundException | DuplicateIdException ex) {
            showError(ex.getMessage());
        }
    }

    private void deleteWorkoutAction() {
        int row = workoutTable.getSelectedRow();
        if (row < 0) {
            showError("Select a workout row to delete.");
            return;
        }
        String logId = workoutTableModel.getValueAt(row, 0).toString();
        try {
            workoutManager.remove(logId);
            saveDataQuietly();
            refreshAllTables();
        } catch (NotFoundException ex) {
            showError(ex.getMessage());
        }
    }

    private String generateWorkoutId() {
        int n = workoutManager.size() + 1;
        while (workoutManager.exists(String.format("W%03d", n))) {
            n++;
        }
        return String.format("W%03d", n);
    }

    private void refreshAllTables() {
        // Reload Trainer dropdown in Members tab
        String prevTrainer = (String) mTrainerBox.getSelectedItem();
        mTrainerBox.removeAllItems();
        mTrainerBox.addItem("NONE - No Trainer");
        for (Trainer t : trainerManager.getAll()) {
            mTrainerBox.addItem(t.getId() + " - " + t.getName());
        }
        if (prevTrainer != null) selectComboByPrefix(mTrainerBox, extractId(prevTrainer));

        // Reload Member dropdowns in Workouts tab
        wMemberBox.removeAllItems();
        wFilterBox.removeAllItems();
        wFilterBox.addItem("ALL - All Members");
        for (Member m : memberManager.getAll()) {
            String item = m.getId() + " - " + m.getName();
            wMemberBox.addItem(item);
            wFilterBox.addItem(item);
        }

        refreshMemberTable(memberManager.getAll());
        refreshTrainerTable();
        refreshWorkoutTable();
        refreshSummaryTab();
    }

    private void refreshMemberTable(ArrayList<Member> members) {
        memberTableModel.setRowCount(0);
        for (Member m : members) {
            memberTableModel.addRow(new Object[]{
                    m.getId(), m.getName(), m.getPhone(), m.getPlan(),
                    String.format("%.1f (%s)", m.calculateBMI(), m.getBmiStatus()),
                    m.getAssignedTrainerName(),
                    String.format("BDT %.0f", m.calculateMonthlyAmount())
            });
        }
    }

    private void refreshTrainerTable() {
        trainerTableModel.setRowCount(0);
        for (Trainer t : trainerManager.getAll()) {
            trainerTableModel.addRow(new Object[]{
                    t.getId(), t.getName(), t.getSpecialty(), t.getExperienceYears(),
                    String.format("BDT %.0f", t.getBaseSalary()),
                    t.getAssignedMembers().size(),
                    String.format("BDT %.0f", t.calculateMonthlyAmount())
            });
        }
    }

    private void refreshWorkoutTable() {
        if (workoutTableModel == null) return;
        String filterId = extractId((String) wFilterBox.getSelectedItem());
        workoutTableModel.setRowCount(0);

        for (WorkoutLog w : workoutManager.getAll()) {
            if ("ALL".equalsIgnoreCase(filterId) || w.getMemberId().equalsIgnoreCase(filterId)) {
                String details = "Strength".equalsIgnoreCase(w.getType())
                        ? w.getSets() + "x" + w.getReps() + " @ " + w.getWeightKg() + " kg"
                        : w.getDurationMinutes() + " min";
                workoutTableModel.addRow(new Object[]{
                        w.getId(), w.getMemberId(), w.getDate(), w.getExerciseName(),
                        w.getType(), details, w.getCaloriesBurned() + " kcal"
                });
            }
        }
    }

    private void refreshSummaryTab() {
        // Demonstrates Runtime Polymorphism using ArrayList<Person>
        ArrayList<Person> allPeople = new ArrayList<>();
        allPeople.addAll(trainerManager.getAll());
        allPeople.addAll(memberManager.getAll());

        double revenue = 0;
        double payroll = 0;
        summaryTableModel.setRowCount(0);

        for (Person p : allPeople) {
            double amount = p.calculateMonthlyAmount();
            if (p instanceof Member) {
                revenue += amount;
            } else if (p instanceof Trainer) {
                payroll += amount;
            }
            summaryTableModel.addRow(new Object[]{
                    p.getId(), p.getName(), p.getClass().getSimpleName(),
                    p.getRole(), String.format("BDT %.0f", amount), p.getSummary()
            });
        }

        summaryStatsLabel.setText(String.format(
                "Members: %d   |   Trainers: %d   |   Workouts: %d   |   Monthly Revenue: BDT %.0f   |   Trainer Payroll: BDT %.0f",
                memberManager.size(), trainerManager.size(), workoutManager.size(), revenue, payroll));
    }

    private void validatePersonFields(String id, String name, String phone, String email)
            throws InvalidDataException {
        if (id == null || id.trim().isEmpty()) {
            throw new InvalidDataException("ID cannot be empty.");
        }
        if (name == null || name.trim().length() < 2) {
            throw new InvalidDataException("Name must be at least 2 characters.");
        }
        if (phone == null || phone.trim().length() < 7) {
            throw new InvalidDataException("Please enter a valid phone number.");
        }
        if (email == null || !email.contains("@")) {
            throw new InvalidDataException("Please enter a valid email address.");
        }
    }

    private Trainer getSelectedTrainerObject() throws NotFoundException {
        String id = extractId((String) mTrainerBox.getSelectedItem());
        if ("NONE".equalsIgnoreCase(id) || id.isEmpty()) {
            return null;
        }
        return trainerManager.findById(id);
    }

    private String extractId(String comboText) {
        if (comboText == null) return "NONE";
        int idx = comboText.indexOf(" - ");
        return idx > 0 ? comboText.substring(0, idx).trim() : comboText.trim();
    }

    private void selectComboByPrefix(JComboBox<String> box, String prefix) {
        for (int i = 0; i < box.getItemCount(); i++) {
            String item = box.getItemAt(i);
            if (item != null && item.toUpperCase().startsWith(prefix.toUpperCase())) {
                box.setSelectedIndex(i);
                return;
            }
        }
    }

    private void showError(String message) {
        JOptionPane.showMessageDialog(this, message, "Input / System Error", JOptionPane.ERROR_MESSAGE);
    }
}
